cmake_minimum_required(VERSION 3.24)
project(neural_selection LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_executable(neural_selection src/main.cu src/live_view.cpp)
set_target_properties(neural_selection PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
target_compile_options(neural_selection PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
target_compile_definitions(neural_selection PRIVATE LIVE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/web")
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
target_link_libraries(neural_selection PRIVATE PNG::PNG Threads::Threads)

add_executable(physiology_protocols tests/physiology.cpp)
target_include_directories(physiology_protocols PRIVATE src)
add_executable(neuromodulation_protocols tests/neuromodulation.cpp)
target_include_directories(neuromodulation_protocols PRIVATE src)
add_executable(pyloric_lp_protocols tests/pyloric_lp.cpp)
target_include_directories(pyloric_lp_protocols PRIVATE src)
add_executable(pyloric_leak_protocols tests/pyloric_leak.cpp)
target_include_directories(pyloric_leak_protocols PRIVATE src)
add_executable(pyloric_pacemaker_protocols tests/pyloric_pacemaker.cpp)
target_include_directories(pyloric_pacemaker_protocols PRIVATE src)
enable_testing()
add_test(NAME physiology_protocols COMMAND physiology_protocols)
add_test(NAME neuromodulation_protocols COMMAND neuromodulation_protocols)
add_test(NAME pyloric_lp_protocols COMMAND pyloric_lp_protocols)
add_test(NAME pyloric_leak_protocols COMMAND pyloric_leak_protocols)
add_test(NAME pyloric_pacemaker_protocols COMMAND pyloric_pacemaker_protocols)
