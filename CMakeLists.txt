cmake_minimum_required(VERSION 3.24)
project(neural_selection LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_executable(neural_selection src/main.cu src/live_view.cpp)
set_target_properties(neural_selection PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
target_compile_options(neural_selection PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
target_compile_definitions(neural_selection PRIVATE LIVE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/web")
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
target_link_libraries(neural_selection PRIVATE PNG::PNG z Threads::Threads)

add_executable(physiology_protocols tests/physiology.cpp)
target_include_directories(physiology_protocols PRIVATE src)
add_executable(neuromodulation_protocols tests/neuromodulation.cpp)
target_include_directories(neuromodulation_protocols PRIVATE src)
add_executable(pyloric_lp_protocols tests/pyloric_lp.cpp)
target_include_directories(pyloric_lp_protocols PRIVATE src)
add_executable(pyloric_leak_protocols tests/pyloric_leak.cpp)
target_include_directories(pyloric_leak_protocols PRIVATE src)
add_executable(pyloric_pacemaker_protocols tests/pyloric_pacemaker.cpp)
target_include_directories(pyloric_pacemaker_protocols PRIVATE src)
add_executable(sensory_relay_protocols tests/sensory_relay.cpp)
target_include_directories(sensory_relay_protocols PRIVATE src)
add_executable(orn_receptor_protocols tests/orn_receptor.cpp)
target_include_directories(orn_receptor_protocols PRIVATE src)
add_executable(aplysia_lfs_protocols tests/aplysia_lfs.cpp)
target_include_directories(aplysia_lfs_protocols PRIVATE src)
add_executable(event_synapses tests/event_synapses.cpp)
target_include_directories(event_synapses PRIVATE src)
add_executable(cuda_relay_parity tests/cuda_relay_parity.cu)
target_include_directories(cuda_relay_parity PRIVATE src)
enable_testing()
add_test(NAME physiology_protocols COMMAND physiology_protocols)
add_test(NAME neuromodulation_protocols COMMAND neuromodulation_protocols)
add_test(NAME pyloric_lp_protocols COMMAND pyloric_lp_protocols)
add_test(NAME pyloric_leak_protocols COMMAND pyloric_leak_protocols)
add_test(NAME pyloric_pacemaker_protocols COMMAND pyloric_pacemaker_protocols)
add_test(NAME sensory_relay_protocols COMMAND sensory_relay_protocols)
add_test(NAME orn_receptor_protocols COMMAND orn_receptor_protocols)
add_test(NAME aplysia_lfs_protocols COMMAND aplysia_lfs_protocols)
add_test(NAME event_synapses COMMAND event_synapses)

# Optional Gymnasium bridge also builds directly with g++ (no CUDA needed).
add_library(gym_brain SHARED gym_adapter/native.cpp)
target_include_directories(gym_brain PRIVATE src)

option(BUILD_GYM_CUDA "Build the optional batched Gymnasium CUDA bridge" OFF)
if(BUILD_GYM_CUDA)
  add_library(gym_brain_gpu SHARED gym_adapter/gpu.cu)
  target_include_directories(gym_brain_gpu PRIVATE src)
  target_compile_options(gym_brain_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true>)
  set_target_properties(gym_brain_gpu PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
endif()
